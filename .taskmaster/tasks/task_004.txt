```typescript
// src/lib/data-analyzer.ts
import { Employee } from './types';

export interface KPIResult {
  totalAgreementRate: number;  // 전체 동의율
  implementedAgreementRate: number;  // 진행인원 중 동의율
  totalCount: number;  // 전체 인원
  agreedCount: number;  // 동의자 수
  implementedCount: number;  // 실시자 수
}

export interface ChartData {
  name: string;
  value: number;
  count: number;
  total: number;
}

export function calculateKPI(employees: Employee[]): KPIResult {
  const totalCount = employees.length;
  const agreedCount = employees.filter(e => e.agreementStatus === 'Y').length;
  const implementedCount = employees.filter(e => e.isImplemented === 'Y').length;
  
  const totalAgreementRate = totalCount > 0 
    ? parseFloat(((agreedCount / totalCount) * 100).toFixed(1)) 
    : 0;
    
  const implementedAgreementRate = implementedCount > 0 
    ? parseFloat(((agreedCount / implementedCount) * 100).toFixed(1)) 
    : 0;
  
  return {
    totalAgreementRate,
    implementedAgreementRate,
    totalCount,
    agreedCount,
    implementedCount
  };
}

export function getDisagreementList(employees: Employee[]): Employee[] {
  return employees.filter(e => e.agreementStatus !== 'Y');
}

export function analyzeByCategory(employees: Employee[], category: string): ChartData[] {
  const result = new Map<string, { agreed: number, total: number }>();
  
  employees.forEach(employee => {
    let key = '';
    
    switch(category) {
      case 'majorOrg':
        key = employee.majorOrg;
        break;
      case 'grade':
        key = employee.grade;
        break;
      case 'gradeYear':
        key = `${employee.gradeYear}년차`;
        break;
      case 'hiringType':
        key = employee.hiringType;
        break;
      case 'position':
        key = employee.position;
        break;
      case 'age':
        const ageGroup = Math.floor(employee.age / 10) * 10;
        key = `${ageGroup}대`;
        break;
      case 'yearsOfService':
        const serviceGroup = Math.floor(employee.yearsOfService / 5) * 5;
        key = `${serviceGroup}~${serviceGroup+4}년`;
        break;
      case 'gender':
        key = employee.gender;
        break;
      default:
        key = '기타';
    }
    
    if (!result.has(key)) {
      result.set(key, { agreed: 0, total: 0 });
    }
    
    const current = result.get(key)!;
    current.total += 1;
    
    if (employee.agreementStatus === 'Y') {
      current.agreed += 1;
    }
  });
  
  // Convert to chart data format
  const chartData: ChartData[] = [];
  
  result.forEach((value, key) => {
    const rate = value.total > 0 
      ? parseFloat(((value.agreed / value.total) * 100).toFixed(1)) 
      : 0;
      
    chartData.push({
      name: key,
      value: rate,
      count: value.agreed,
      total: value.total
    });
  });
  
  // Sort by name
  return chartData.sort((a, b) => a.name.localeCompare(b.name));
}
```
