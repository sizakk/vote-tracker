```typescript
// src/lib/types.ts
export interface Employee {
  employeeId: string;    // 사번
  name: string;          // 성명
  majorOrg: string;      // 대조직
  organization: string;  // 조직
  grade: string;         // Grade
  gradeYear: number;     // Grade년차
  position: string;      // 직책
  hiringType: string;    // 입사구분
  gender: string;        // 성별
  age: number;           // 나이
  yearsOfService: number; // 근속년수
  agreementStatus: string; // 동의여부 (Y/N)
  isImplemented: string;   // 실시여부 (Y/N)
}

// src/lib/excel-parser.ts
import * as XLSX from 'xlsx';
import { Employee } from './types';

export class ExcelParseError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'ExcelParseError';
  }
}

export async function parseExcelFile(file: File): Promise<Employee[]> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target?.result as ArrayBuffer);
        const workbook = XLSX.read(data, { type: 'array' });
        
        // Get first sheet
        const worksheet = workbook.Sheets[workbook.SheetNames[0]];
        
        // Convert to JSON
        const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1 }) as any[][];
        
        // Validate column count
        if (jsonData[0].length < 19) {
          throw new ExcelParseError('Excel 파일의 열 구조가 올바르지 않습니다. 19개의 열이 필요합니다.');
        }
        
        // Skip header row and map data to Employee interface
        const employees: Employee[] = [];
        
        for (let i = 1; i < jsonData.length; i++) {
          const row = jsonData[i];
          
          // Skip empty rows
          if (!row[0]) continue;
          
          const employee: Employee = {
            employeeId: String(row[0] || ''),
            name: String(row[1] || ''),
            majorOrg: String(row[2] || ''),
            organization: String(row[3] || ''),
            grade: String(row[4] || ''),
            gradeYear: Number(row[5] || 0),
            position: String(row[6] || ''),
            hiringType: String(row[7] || ''),
            gender: String(row[8] || ''),
            age: Number(row[9] || 0),
            yearsOfService: Number(row[10] || 0),
            agreementStatus: String(row[11] || 'N'),
            isImplemented: String(row[12] || 'N')
          };
          
          employees.push(employee);
        }
        
        resolve(employees);
      } catch (error) {
        if (error instanceof ExcelParseError) {
          reject(error);
        } else {
          reject(new ExcelParseError('Excel 파일 파싱 중 오류가 발생했습니다.'));
        }
      }
    };
    
    reader.onerror = () => {
      reject(new ExcelParseError('파일 읽기 오류가 발생했습니다.'));
    };
    
    reader.readAsArrayBuffer(file);
  });
}
```
