```typescript
// src/components/KPICards.tsx
import { Card, CardContent, CardHeader, CardTitle } from './ui/card';
import { PieChart, Pie, Cell, ResponsiveContainer, Tooltip } from 'recharts';
import { KPIResult } from '@/lib/data-analyzer';

interface KPICardsProps {
  kpiData: KPIResult;
}

export default function KPICards({ kpiData }: KPICardsProps) {
  const { totalAgreementRate, implementedAgreementRate, totalCount, agreedCount, implementedCount } = kpiData;
  
  // Data for total agreement rate pie chart
  const totalData = [
    { name: '동의', value: agreedCount },
    { name: '미동의', value: totalCount - agreedCount }
  ];
  
  // Data for implemented agreement rate pie chart
  const implementedData = [
    { name: '동의', value: agreedCount },
    { name: '미동의', value: implementedCount - agreedCount }
  ];
  
  const COLORS = ['#4CAF50', '#F44336'];
  
  return (
    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
      <Card>
        <CardHeader>
          <CardTitle>전체 동의율</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col items-center">
          <div className="text-3xl font-bold mb-2">{totalAgreementRate}%</div>
          <div className="text-sm text-gray-500 mb-4">{agreedCount} / {totalCount}명</div>
          <div className="w-full h-48">
            <ResponsiveContainer width="100%" height="100%">
              <PieChart>
                <Pie
                  data={totalData}
                  cx="50%"
                  cy="50%"
                  innerRadius={60}
                  outerRadius={80}
                  paddingAngle={5}
                  dataKey="value"
                >
                  {totalData.map((entry, index) => (
                    <Cell key={`cell-${index}`} fill={COLORS[index % COLORS.length]} />
                  ))}
                </Pie>
                <Tooltip formatter={(value) => [`${value}명`, '']} />
              </PieChart>
            </ResponsiveContainer>
          </div>
        </CardContent>
      </Card>
      
      <Card>
        <CardHeader>
          <CardTitle>진행인원 중 동의율</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col items-center">
          <div className="text-3xl font-bold mb-2">{implementedAgreementRate}%</div>
          <div className="text-sm text-gray-500 mb-4">{agreedCount} / {implementedCount}명</div>
          <div className="w-full h-48">
            <ResponsiveContainer width="100%" height="100%">
              <PieChart>
                <Pie
                  data={implementedData}
                  cx="50%"
                  cy="50%"
                  innerRadius={60}
                  outerRadius={80}
                  paddingAngle={5}
                  dataKey="value"
                >
                  {implementedData.map((entry, index) => (
                    <Cell key={`cell-${index}`} fill={COLORS[index % COLORS.length]} />
                  ))}
                </Pie>
                <Tooltip formatter={(value) => [`${value}명`, '']} />
              </PieChart>
            </ResponsiveContainer>
          </div>
        </CardContent>
      </Card>
    </div>
  );
}
```

# Test Strategy:
1. 다양한 KPI 값으로 렌더링 테스트
2. 원형 차트가 데이터에 따라 올바르게 표시되는지 확인
3. 반응형 레이아웃 테스트 (모바일, 태블릿, 데스크톱)
4. 엣지 케이스 테스트 (0%, 100%, 데이터 없음 등)
5. 접근성 테스트 (스크린 리더 호환성)
