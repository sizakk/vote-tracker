```typescript
// src/components/ResultChart.tsx
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend, ResponsiveContainer } from 'recharts';
import { ChartData } from '@/lib/data-analyzer';
import { Card, CardContent, CardHeader, CardTitle } from './ui/card';

interface ResultChartProps {
  data: ChartData[];
  category: string;
}

interface CustomTooltipProps {
  active?: boolean;
  payload?: any[];
}

const CustomTooltip = ({ active, payload }: CustomTooltipProps) => {
  if (active && payload && payload.length) {
    const data = payload[0].payload;
    return (
      <div className="bg-white p-3 border rounded shadow-sm">
        <p className="font-semibold">{data.name}</p>
        <p>동의율: {data.value}%</p>
        <p>동의: {data.count}명 / 전체: {data.total}명</p>
      </div>
    );
  }
  return null;
};

const getCategoryTitle = (category: string): string => {
  const titles: Record<string, string> = {
    majorOrg: '대조직별 동의율',
    grade: 'Grade별 동의율',
    gradeYear: 'Grade년차별 동의율',
    hiringType: '입사구분별 동의율',
    position: '직책별 동의율',
    age: '나이대별 동의율',
    yearsOfService: '근속년수별 동의율',
    gender: '성별별 동의율',
    department: '구분별 동의율'
  };
  
  return titles[category] || '분석 결과';
};

export default function ResultChart({ data, category }: ResultChartProps) {
  if (!data || data.length === 0) {
    return (
      <Card>
        <CardContent className="pt-6 text-center">
          <p>데이터가 없습니다.</p>
        </CardContent>
      </Card>
    );
  }
  
  return (
    <Card>
      <CardHeader>
        <CardTitle>{getCategoryTitle(category)}</CardTitle>
      </CardHeader>
      <CardContent>
        <div className="w-full h-[400px]">
          <ResponsiveContainer width="100%" height="100%">
            <BarChart
              data={data}
              margin={{ top: 20, right: 30, left: 20, bottom: 70 }}
            >
              <CartesianGrid strokeDasharray="3 3" />
              <XAxis 
                dataKey="name" 
                angle={-45} 
                textAnchor="end" 
                height={70} 
              />
              <YAxis 
                label={{ value: '동의율 (%)', angle: -90, position: 'insideLeft' }} 
                domain={[0, 100]} 
              />
              <Tooltip content={<CustomTooltip />} />
              <Legend />
              <Bar dataKey="value" name="동의율 (%)" fill="#4CAF50" />
            </BarChart>
          </ResponsiveContainer>
        </div>
      </CardContent>
    </Card>
  );
}
```
