```typescript
// src/components/AnalysisButtons.tsx
import { Button } from './ui/button';

interface AnalysisButtonsProps {
  onSelectAnalysis: (category: string) => void;
  selectedCategory: string | null;
}

export default function AnalysisButtons({ onSelectAnalysis, selectedCategory }: AnalysisButtonsProps) {
  const analysisCategories = [
    { id: 'majorOrg', label: '대조직별' },
    { id: 'grade', label: 'Grade별' },
    { id: 'gradeYear', label: 'Grade년차별' },
    { id: 'hiringType', label: '입사구분별' },
    { id: 'position', label: '직책별' },
    { id: 'age', label: '나이대별' },
    { id: 'yearsOfService', label: '근속년수별' },
    { id: 'gender', label: '성별별' },
    { id: 'department', label: '구분별' }
  ];
  
  return (
    <div className="my-6">
      <h2 className="text-lg font-semibold mb-3">분석 카테고리</h2>
      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-2">
        {analysisCategories.map((category) => (
          <Button
            key={category.id}
            variant={selectedCategory === category.id ? "default" : "outline"}
            onClick={() => onSelectAnalysis(category.id)}
            className="w-full"
          >
            {category.label}
          </Button>
        ))}
      </div>
    </div>
  );
}
```
